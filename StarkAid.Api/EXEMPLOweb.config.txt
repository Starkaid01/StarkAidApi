<?xml version="1.0" encoding="utf-8"?>
<configuration>
  <system.webServer>

    <!-- Configura o ASP.NET Core Module para executar a aplicação -->
    <handlers>
      <add name="aspNetCore" path="*" verb="*" modules="AspNetCoreModuleV2" resourceType="Unspecified"/>
    </handlers>

    <!-- Protege arquivos de configuração e sensíveis -->
    <security>
      <requestFiltering>
        <hiddenSegments>
          <add segment="appsettings.json" />
          <add segment="appsettings.Production.json" />
          <add segment="appsettings.Development.json" />
          <add segment="appsettings.Staging.json" />
          <add segment="secrets.json" />
        </hiddenSegments>
      </requestFiltering>
    </security>

  </system.webServer>

  <!-- Configura o módulo ASP.NET Core -->
  <system.webServer>
    <aspNetCore processPath="dotnet" arguments="StarkAid.Api.dll" stdoutLogEnabled="true" stdoutLogFile=".\logs\stdout" hostingModel="inprocess"/>
  </system.webServer>

</configuration>
